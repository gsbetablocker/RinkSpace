cmake_minimum_required(VERSION 3.20)
project(rinkspace_core LANGUAGES CXX)
option(RINKSPACE_ENABLE_CUDA "Build the resident CUDA evaluator/linearization" OFF)
option(RINKSPACE_ENABLE_OPENMP "Parallelize independent CPU motion-projection rows" OFF)
set(RINKSPACE_CUDSS_ROOT "" CACHE PATH "Existing cuDSS root for the optional native LM candidate")
add_library(rinkspace_core SHARED src/core_api.cpp)
# Some localized MSVC/Ninja combinations misdecode /showIncludes and record
# zero header dependencies. Keep project-header changes explicit so CPU and
# CUDA cannot silently use different revisions of the shared mathematics.
file(GLOB_RECURSE RINKSPACE_PUBLIC_HEADERS CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/include/rinkspace/*.h"
  "${CMAKE_CURRENT_SOURCE_DIR}/include/rinkspace/*.hpp")
set_property(SOURCE src/core_api.cpp APPEND PROPERTY OBJECT_DEPENDS "${RINKSPACE_PUBLIC_HEADERS}")
target_include_directories(rinkspace_core PUBLIC include)
target_compile_features(rinkspace_core PUBLIC cxx_std_17)
target_compile_definitions(rinkspace_core PRIVATE RINKSPACE_CORE_BUILD)
if(RINKSPACE_ENABLE_OPENMP)
  find_package(OpenMP COMPONENTS CXX QUIET)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(rinkspace_core PRIVATE OpenMP::OpenMP_CXX)
  else()
    message(STATUS "OpenMP CXX unavailable: motion projection remains serial and reports this capability")
  endif()
endif()
set_target_properties(rinkspace_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_compile_options(rinkspace_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/fp:precise>)
else()
  target_compile_options(rinkspace_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endif()
if(RINKSPACE_ENABLE_CUDA)
  enable_language(CUDA)
  if(NOT CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES native)
  endif()
  set_property(TARGET rinkspace_core PROPERTY CUDA_ARCHITECTURES ${CMAKE_CUDA_ARCHITECTURES})
  target_sources(rinkspace_core PRIVATE src/core_gpu.cu)
  if(RINKSPACE_CUDSS_ROOT)
    find_package(CUDAToolkit REQUIRED)
    find_path(RINKSPACE_CUDSS_INCLUDE cudss.h PATHS "${RINKSPACE_CUDSS_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
    find_library(RINKSPACE_CUDSS_LIBRARY cudss PATHS "${RINKSPACE_CUDSS_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    target_include_directories(rinkspace_core PRIVATE "${RINKSPACE_CUDSS_INCLUDE}")
    target_compile_definitions(rinkspace_core PRIVATE RINKSPACE_HAVE_CUDSS)
    target_link_libraries(rinkspace_core PRIVATE CUDA::cusolver "${RINKSPACE_CUDSS_LIBRARY}")
  endif()

  set_target_properties(rinkspace_core PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  set(RINKSPACE_CUDA_FAST_COMPILE "0" CACHE STRING "CUDA device optimizer compile-time tradeoff: 0/min/mid/max")
  if(NOT RINKSPACE_CUDA_FAST_COMPILE STREQUAL "0")
    target_compile_options(rinkspace_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--Ofast-compile=${RINKSPACE_CUDA_FAST_COMPILE}>)
  endif()
  target_compile_options(rinkspace_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr;--fmad=false;--diag-error=20011;--diag-error=20014>)
endif()
include(CTest)
if(BUILD_TESTING)
  set_property(SOURCE tests/coupled_dual.cpp APPEND PROPERTY OBJECT_DEPENDS "${RINKSPACE_PUBLIC_HEADERS}")
  add_executable(rinkspace_coupled_dual tests/coupled_dual.cpp)
  target_include_directories(rinkspace_coupled_dual PRIVATE include)
  target_compile_features(rinkspace_coupled_dual PRIVATE cxx_std_17)
  add_test(NAME rinkspace_coupled_dual COMMAND rinkspace_coupled_dual)
  set_property(SOURCE tests/coupled_dual_nonfinite.cpp APPEND PROPERTY OBJECT_DEPENDS "${RINKSPACE_PUBLIC_HEADERS}")
  add_executable(rinkspace_coupled_dual_nonfinite tests/coupled_dual_nonfinite.cpp)
  target_include_directories(rinkspace_coupled_dual_nonfinite PRIVATE include)
  target_compile_features(rinkspace_coupled_dual_nonfinite PRIVATE cxx_std_17)
  add_test(NAME rinkspace_coupled_dual_nonfinite COMMAND rinkspace_coupled_dual_nonfinite)
  set_property(SOURCE tests/coupled_dual_working_set.cpp APPEND PROPERTY OBJECT_DEPENDS "${RINKSPACE_PUBLIC_HEADERS}")
  add_executable(rinkspace_coupled_dual_working_set tests/coupled_dual_working_set.cpp)
  target_include_directories(rinkspace_coupled_dual_working_set PRIVATE include)
  target_compile_features(rinkspace_coupled_dual_working_set PRIVATE cxx_std_17)
  add_test(NAME rinkspace_coupled_dual_working_set COMMAND rinkspace_coupled_dual_working_set)
  set_property(SOURCE tests/ice_tolerance.cpp APPEND PROPERTY OBJECT_DEPENDS "${RINKSPACE_PUBLIC_HEADERS}")
  add_executable(rinkspace_ice_tolerance tests/ice_tolerance.cpp)
  target_include_directories(rinkspace_ice_tolerance PRIVATE include)
  target_compile_features(rinkspace_ice_tolerance PRIVATE cxx_std_17)
  add_test(NAME rinkspace_ice_tolerance COMMAND rinkspace_ice_tolerance)
  foreach(ice_check ice_relation_meta ice_actual_witness ice_actual_chart ice_witness_boundary)
    add_executable(rinkspace_${ice_check} tests/${ice_check}.cpp)
    target_include_directories(rinkspace_${ice_check} PRIVATE include)
    target_compile_features(rinkspace_${ice_check} PRIVATE cxx_std_17)
    set_property(SOURCE tests/${ice_check}.cpp APPEND PROPERTY OBJECT_DEPENDS "${RINKSPACE_PUBLIC_HEADERS}")
    add_test(NAME rinkspace_${ice_check} COMMAND rinkspace_${ice_check})
  endforeach()
  add_executable(rinkspace_core_smoke tests/core_smoke.cpp)
  target_link_libraries(rinkspace_core_smoke PRIVATE rinkspace_core)
  target_compile_features(rinkspace_core_smoke PRIVATE cxx_std_17)
  add_test(NAME rinkspace_core_smoke COMMAND rinkspace_core_smoke)
  add_executable(rinkspace_box_stationarity tests/box_stationarity.cpp)
  target_include_directories(rinkspace_box_stationarity PRIVATE include)
  target_compile_features(rinkspace_box_stationarity PRIVATE cxx_std_17)
  add_test(NAME rinkspace_box_stationarity COMMAND rinkspace_box_stationarity)
  if(RINKSPACE_ENABLE_CUDA)
    add_executable(rinkspace_attach_scene_cuda tests/attach_scene_cuda.cu)
    target_include_directories(rinkspace_attach_scene_cuda PRIVATE include)
    target_compile_definitions(rinkspace_attach_scene_cuda PRIVATE RINKSPACE_CORE_BUILD)
    set_target_properties(rinkspace_attach_scene_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
    target_compile_options(rinkspace_attach_scene_cuda PRIVATE
      $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr;--fmad=false;--diag-error=20011;--diag-error=20014>)
    if(RINKSPACE_CUDSS_ROOT)
      target_compile_definitions(rinkspace_attach_scene_cuda PRIVATE RINKSPACE_HAVE_CUDSS)
      target_include_directories(rinkspace_attach_scene_cuda PRIVATE "${RINKSPACE_CUDSS_INCLUDE}")
      target_link_libraries(rinkspace_attach_scene_cuda PRIVATE CUDA::cusolver "${RINKSPACE_CUDSS_LIBRARY}")
    endif()
    add_test(NAME rinkspace_attach_scene_cuda COMMAND rinkspace_attach_scene_cuda)
    add_executable(rinkspace_g2_status_cuda tests/g2_status_cuda.cu)
    target_include_directories(rinkspace_g2_status_cuda PRIVATE include)
    target_compile_definitions(rinkspace_g2_status_cuda PRIVATE RINKSPACE_CORE_BUILD)
    set_property(SOURCE tests/g2_status_cuda.cu APPEND PROPERTY OBJECT_DEPENDS
      "${RINKSPACE_PUBLIC_HEADERS};${CMAKE_CURRENT_SOURCE_DIR}/src/core_gpu.cu")
    set_target_properties(rinkspace_g2_status_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
    target_compile_options(rinkspace_g2_status_cuda PRIVATE
      $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr;--fmad=false;--diag-error=20011;--diag-error=20014>)
    add_test(NAME rinkspace_g2_status_cuda COMMAND rinkspace_g2_status_cuda)
    set_tests_properties(rinkspace_g2_status_cuda PROPERTIES SKIP_RETURN_CODE 77)

  endif()
endif()
