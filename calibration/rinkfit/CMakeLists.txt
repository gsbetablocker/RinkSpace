cmake_minimum_required(VERSION 3.20)
project(rinkfit_start LANGUAGES C CXX)

option(RINKFIT_BUILD_PRODUCTION_BACKEND "Build Ceres/SuiteSparse AL backend and synthetic fixture" ON)
include(CTest)

add_library(rinkfit_al STATIC cpp/src/augmented_lagrangian.cpp cpp/src/checkpoint.cpp)
target_include_directories(rinkfit_al PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/cpp/include)
target_compile_features(rinkfit_al PUBLIC cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(rinkfit_al PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_TESTING)
  add_executable(test_augmented_lagrangian cpp/tests/test_augmented_lagrangian.cpp)
  target_link_libraries(test_augmented_lagrangian PRIVATE rinkfit_al)
  add_test(NAME augmented_lagrangian_reference COMMAND test_augmented_lagrangian)
  add_executable(test_physical cpp/tests/test_physical.cpp)
  target_link_libraries(test_physical PRIVATE rinkfit_al)
  add_test(NAME physical_reference COMMAND test_physical)
endif()

if(RINKFIT_BUILD_PRODUCTION_BACKEND)
  find_package(Ceres 2.2 QUIET COMPONENTS SuiteSparse)
  if(NOT Ceres_FOUND)
    message(FATAL_ERROR
      "Missing production dependency: Ceres >= 2.2 built with SuiteSparse. "
      "Install Ceres development files, Eigen, SuiteSparse and CMake; "
      "set Ceres_DIR if needed. No SciPy/dense substitute is selected. "
      "For independent standard-C++ mathematical tests only, configure "
      "-DRINKFIT_BUILD_PRODUCTION_BACKEND=OFF (this does not build a fitter).")
  endif()
  add_library(rinkfit_ceres STATIC cpp/src/ceres_backend.cpp)
  target_link_libraries(rinkfit_ceres PUBLIC rinkfit_al Ceres::ceres)
  target_compile_features(rinkfit_ceres PUBLIC cxx_std_17)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(rinkfit_ceres PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_executable(rinkfit_synthetic_demo cpp/src/synthetic_demo.cpp)
  target_link_libraries(rinkfit_synthetic_demo PRIVATE rinkfit_ceres)
  if(BUILD_TESTING)
    add_test(NAME ceres_finite_material_and_resume COMMAND rinkfit_synthetic_demo --self-test)
  endif()
  find_package(nlohmann_json CONFIG REQUIRED)
  add_library(rinkfit_event STATIC cpp/src/event_model.cpp cpp/src/event_builder.cpp)
  target_link_libraries(rinkfit_event PUBLIC rinkfit_ceres nlohmann_json::nlohmann_json)
  add_executable(rinkfit_event_fit cpp/src/event_fit.cpp)
  target_link_libraries(rinkfit_event_fit PRIVATE rinkfit_event)
  if(MSVC)
    target_compile_options(rinkfit_event PRIVATE /utf-8 /bigobj)
    target_compile_options(rinkfit_event_fit PRIVATE /utf-8 /bigobj)
  endif()
  add_library(rinkfit_joint STATIC cpp/src/joint_model.cpp cpp/src/joint_packet.cpp cpp/src/joint_builder.cpp cpp/src/joint_capture.cpp cpp/src/joint_al.cpp cpp/src/joint_gpu_cache.cpp cpp/src/joint_gpu_driver.cpp cpp/src/joint_recovery.cpp)
  target_link_libraries(rinkfit_joint PUBLIC rinkfit_event)
  add_executable(rinkfit_joint_fit cpp/src/joint_fit.cpp)
  target_link_libraries(rinkfit_joint_fit PRIVATE rinkfit_joint)
  if(MSVC)
    target_compile_options(rinkfit_joint PRIVATE /utf-8 /bigobj)
    target_compile_options(rinkfit_joint_fit PRIVATE /utf-8 /bigobj)
  endif()
  # Full native UV and original-W transport for the reviewed shot ABI.
  # These tools evaluate existing physical states; they do not claim a final fit.
  add_executable(rinkfit_reduced_qbar tools/reduced/transport_qbar_base.cpp)
  target_link_libraries(rinkfit_reduced_qbar PRIVATE rinkfit_event)
  add_executable(rinkfit_native_uv tools/reduced/whole_W_transport.cpp)
  target_link_libraries(rinkfit_native_uv PRIVATE rinkfit_event)
  if(MSVC)
    target_compile_options(rinkfit_reduced_qbar PRIVATE /utf-8 /bigobj)
    target_compile_options(rinkfit_native_uv PRIVATE /utf-8 /bigobj)
  endif()
  if(BUILD_TESTING)
    add_executable(test_event_model cpp/tests/test_event_model.cpp)
    target_link_libraries(test_event_model PRIVATE rinkfit_event)
    add_test(NAME event_physical_model COMMAND test_event_model)
    add_executable(test_event_factors cpp/tests/test_event_factors.cpp)
    target_link_libraries(test_event_factors PRIVATE rinkfit_event)
    add_test(NAME event_actual_derivatives COMMAND test_event_factors)
    add_executable(test_phr_kink cpp/tests/test_phr_kink.cpp)
    target_link_libraries(test_phr_kink PRIVATE rinkfit_ceres nlohmann_json::nlohmann_json)
    add_test(NAME phr_kink_same_objective COMMAND test_phr_kink 10000 ${CMAKE_CURRENT_BINARY_DIR}/kink_report.json)
    add_executable(test_event_packet cpp/tests/test_event_packet.cpp)
    target_link_libraries(test_event_packet PRIVATE rinkfit_al nlohmann_json::nlohmann_json)
    if(MSVC)
      target_compile_options(test_event_model PRIVATE /UNDEBUG)
      target_compile_options(test_event_factors PRIVATE /UNDEBUG)
      target_compile_options(test_event_packet PRIVATE /UNDEBUG /utf-8)
    else()
      target_compile_options(test_event_model PRIVATE -UNDEBUG)
      target_compile_options(test_event_factors PRIVATE -UNDEBUG)
      target_compile_options(test_event_packet PRIVATE -UNDEBUG)
    endif()
  endif()
endif()
